# The Map Table Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `dnd_map.py` with two browser pages — a DM page that paints fog of war over a map, and a projector page locked to the same view.

**Architecture:** `assets/fog.js` holds every pure function (view maths, the stroke model, the rasteriser) so both pages share one implementation and one test suite. Both pages are same-origin and share IndexedDB, so the map image never travels between them — `BroadcastChannel` carries only a view, a stroke, or a map id. Fog is a list of strokes rasterised into an alpha mask, which is what buys undo.

**Tech Stack:** Plain HTML/CSS/JS, Canvas 2D, IndexedDB, BroadcastChannel. No build step, no dependencies.

**Spec:** `docs/superpowers/specs/2026-08-11-map-table-design.md`

## Global Constraints

- **No build step, no package manager, no CLI test runner.** Tests are in-page at `?test=1`, the convention `editor.html` established. Serve with `python -m http.server 8731 --bind 127.0.0.1`.
- **Page CSS stays inline in the page.** Do not touch `assets/wazoo.css` — changing an already-shared asset forces bumping `?v=` in every page that references it.
- **`assets/fog.js` is new** and referenced only by the two new pages, so it needs no bump elsewhere. Reference it as `assets/fog.js?v=20260808n`, the current token.
- **Neither page loads `gate.js` or `keyring.js`.** No ward. The spec's reasoning: this tool writes nothing to the repository, so a ward protects nothing and a forgotten token would leave the DM with no map mid-session.
- **The repository is PUBLIC.** No map images, no fog data, no DM notes are ever committed. Everything this tool stores lives in IndexedDB on the DM's machine.
- **Both pages get `<meta name="robots" content="noindex, nofollow">`** and must be added to `HIDDEN` in `press-room.html`.
- **The mask is an alpha mask**: opaque black where hidden, transparent where revealed. It is drawn straight over the map — at `0.78` alpha on the DM page (matching `EDITOR_FOG_ALPHA` in `dnd_map.py`), fully opaque on the projector.
- **All stroke coordinates are image-space**, never canvas-space. That is what makes two differently-sized windows produce identical masks.

```javascript
/* ═══════════════════════════════════════════════════════════════════
   Fog — the map table's arithmetic, shared by both windows.

   Everything here is pure: no DOM beyond an offscreen canvas, no
   storage, no channel state. The DM page and the projector page both
   load it, which is how they are guaranteed to agree.

   Two things are worth knowing before changing anything:

   - Stroke coordinates are ALWAYS in image space. The two windows have
     different canvas sizes and probably different aspect ratios; only
     image space makes their masks identical.
   - The mask is an ALPHA mask, not a black-and-white picture. It is
     opaque where the map is hidden and transparent where it has been
     revealed, so it can be drawn straight over the map with no
     compositing tricks at the call site.
   ═══════════════════════════════════════════════════════════════════ */
(function (root) {
  'use strict';

  var CHANNEL = 'waterdeep-table';

  /* ── view maths ───────────────────────────────────────────────────
     A view is {z, tx, ty}: scale about the origin, then translate.
     canvas = image * z + t                                            */
  function imageToCanvas(v, ix, iy){ return { x: ix * v.z + v.tx, y: iy * v.z + v.ty }; }
  function canvasToImage(v, cx, cy){ return { x: (cx - v.tx) / v.z, y: (cy - v.ty) / v.z }; }

  function fitScale(w, h, cw, ch){
    if(!w || !h) return 1;
    return Math.min(cw / w, ch / h);
  }

  function fitView(w, h, cw, ch){
    var z = fitScale(w, h, cw, ch);
    return { z: z, tx: (cw - w * z) / 2, ty: (ch - h * z) / 2 };
  }

  /* Zoom about a canvas point, keeping whatever is under it still.
     The image point under the cursor must not move, so the translation
     has to absorb the scale change. */
  function zoomAt(v, cx, cy, factor, min, max){
    var z = Math.min(max, Math.max(min, v.z * factor));
    var k = z / v.z;
    return { z: z, tx: cx - (cx - v.tx) * k, ty: cy - (cy - v.ty) * k };
  }

  /* ── strokes ──────────────────────────────────────────────────────── */
  function line(x0,y0,x1,y1,r,reveal){
    return {k:"line", x0:x0, y0:y0, x1:x1, y1:y1, r:r, reveal:!!reveal};
  }
  function rect(x0,y0,x1,y1,reveal){
    return {k:"rect", x0:x0, y0:y0, x1:x1, y1:y1, reveal:!!reveal};
  }
  function fill(reveal){ return {k:"fill", reveal:!!reveal}; }

  /* Draw the strokes, in order, into an alpha mask of w×h.
     Starts fully fogged; a reveal erases, a hide paints back. */
  function rasterise(strokes, w, h, canvas){
    var cv = canvas || document.createElement("canvas");
    if(cv.width !== w) cv.width = w;
    if(cv.height !== h) cv.height = h;
    var g = cv.getContext("2d");

    g.setTransform(1,0,0,1,0,0);
    g.globalCompositeOperation = "source-over";
    g.clearRect(0, 0, w, h);
    g.fillStyle = "#000";
    g.fillRect(0, 0, w, h);

    (strokes || []).forEach(function(s){
      /* destination-out erases what is already there — that is a reveal.
         source-over paints opaque black back — that is a hide. */
      g.globalCompositeOperation = s.reveal ? "destination-out" : "source-over";
      g.fillStyle = "#000";
      g.strokeStyle = "#000";

      if(s.k === "fill"){ g.fillRect(0, 0, w, h); return; }

      if(s.k === "rect"){
        g.fillRect(Math.min(s.x0, s.x1), Math.min(s.y0, s.y1),
                   Math.abs(s.x1 - s.x0), Math.abs(s.y1 - s.y0));
        return;
      }

      if(s.k === "line"){
        /* A click is a zero-length line, and stroking one draws nothing
           in every browser worth naming. Paint the cap by hand. */
        if(s.x0 === s.x1 && s.y0 === s.y1){
          g.beginPath(); g.arc(s.x0, s.y0, s.r, 0, Math.PI * 2); g.fill();
          return;
        }
        g.lineWidth = s.r * 2;
        g.lineCap = "round";
        g.lineJoin = "round";
        g.beginPath(); g.moveTo(s.x0, s.y0); g.lineTo(s.x1, s.y1); g.stroke();
      }
    });

    g.globalCompositeOperation = "source-over";
    return cv;
  }

  /* Rotate strokes 90° clockwise for an image that was `oldH` tall.
     Rotating the picture and leaving the fog behind is the bug this
     prevents, so both are transformed together — as dnd_map.py does. */
  function rotateStrokes(strokes, oldH){
    return (strokes || []).map(function(s){
      if(s.k === "fill") return s;
      var o = {k:s.k, reveal:s.reveal};
      o.x0 = oldH - s.y0; o.y0 = s.x0;
      o.x1 = oldH - s.y1; o.y1 = s.x1;
      if(s.k === "line") o.r = s.r;
      return o;
    });
  }

  /* Turn an image 90° clockwise, `times` times, into a fresh canvas.
     The stored blob is never rewritten, so the orientation has to be
     re-applied every time the map is decoded — on this page after a
     reload, and on the projector, which fetches the same blob itself.
     Rotating the strokes but not the picture is the misalignment this
     exists to prevent. */
  function rotateImage(img, times){
    var n = ((times % 4) + 4) % 4;
    if(!n) return img;
    var w = img.width, h = img.height;
    var out = img;
    for(var i = 0; i < n; i++){
      var cv = document.createElement("canvas");
      cv.width = h; cv.height = w;
      var g = cv.getContext("2d");
      g.translate(h, 0);
      g.rotate(Math.PI / 2);
      g.drawImage(out, 0, 0);
      out = cv;
      var t = w; w = h; h = t;
    }
    return out;
  }

  /* ── the channel ──────────────────────────────────────────────────── */
  function open(onMessage){
    if(typeof BroadcastChannel !== "function") return null;
    var ch = new BroadcastChannel(CHANNEL);
    ch.onmessage = function(e){ onMessage(e.data); };
    return ch;
  }

  root.Fog = {
    channelName: CHANNEL,
    imageToCanvas: imageToCanvas,
    canvasToImage: canvasToImage,
    fitScale: fitScale,
    fitView: fitView,
    zoomAt: zoomAt,
    line: line,
    rect: rect,
    fill: fill,
    rasterise: rasterise,
    rotateStrokes: rotateStrokes,
    rotateImage: rotateImage,
    open: open
  };
})(window);
```

```javascript
/* =========================================================================
   The table.

   Both this page and the projector are the same origin, so they share
   IndexedDB. The map is stored once, here; the projector is told only
   its id and fetches the blob itself. A 15 MB image never crosses the
   channel.
   ========================================================================= */
const DB = "waterdeep-table", MAPS = "maps", FOG = "fog";
const FOG_ALPHA = 0.78;          /* EDITOR_FOG_ALPHA in dnd_map.py */
const ZMIN = 0.02, ZMAX = 40;

let idbOK = true;

function idb(){
  return new Promise(function(res, rej){
    const r = indexedDB.open(DB, 1);
    r.onupgradeneeded = function(){
      const db = r.result;
      if(!db.objectStoreNames.contains(MAPS)) db.createObjectStore(MAPS, {keyPath:"id"});
      if(!db.objectStoreNames.contains(FOG))  db.createObjectStore(FOG,  {keyPath:"mapId"});
    };
    r.onsuccess = function(){ res(r.result); };
    r.onerror = function(){ rej(r.error); };
  });
}

function idbPut(store, rec){
  return idb().then(function(db){
    return new Promise(function(res, rej){
      const tx = db.transaction(store, "readwrite");
      tx.objectStore(store).put(rec);
      tx.oncomplete = res; tx.onerror = function(){ rej(tx.error); };
    });
  });
}

function idbGet(store, key){
  return idb().then(function(db){
    return new Promise(function(res, rej){
      const tx = db.transaction(store, "readonly");
      const rq = tx.objectStore(store).get(key);
      rq.onsuccess = function(){ res(rq.result || null); };
      rq.onerror = function(){ rej(rq.error); };
    });
  });
}

const putMap = (id, name, blob, w, h) => idbPut(MAPS, {id:id, name:name, blob:blob, w:w, h:h});
const getMap = id => idbGet(MAPS, id);
/* Rotation is stored beside the strokes, never baked into the blob. The
   strokes are saved already rotated, so the picture has to be turned by
   the same amount every time it is decoded or the fog lands crooked. */
const putFog = (mapId, strokes, rotation) => idbPut(FOG, {mapId:mapId, strokes:strokes, rotation:rotation||0});
const getFog = mapId => idbGet(FOG, mapId);

const state = { mapId:null, name:"", img:null, w:0, h:0, rotation:0, strokes:[], view:{z:1,tx:0,ty:0} };
let mask = document.createElement("canvas");

function emptyMessage(){ return "No map on the table"; }

function say(msg){ document.getElementById("said").textContent = msg || ""; }

function sizeCanvas(){
  const cv = document.getElementById("table");
  const dpr = window.devicePixelRatio || 1;
  cv.width  = Math.round(cv.clientWidth  * dpr);
  cv.height = Math.round(cv.clientHeight * dpr);
  return cv;
}

function redraw(){
  const cv = document.getElementById("table");
  const g = cv.getContext("2d");
  g.setTransform(1,0,0,1,0,0);
  g.fillStyle = "#0b0805";
  g.fillRect(0,0,cv.width,cv.height);

  if(!state.img){
    g.fillStyle = "#6d5a3a";
    g.font = '16px "EB Garamond", Georgia, serif';
    g.textAlign = "center";
    g.fillText(emptyMessage(), cv.width/2, cv.height/2);
    return;
  }

  const v = state.view;
  g.imageSmoothingQuality = "high";
  g.drawImage(state.img, v.tx, v.ty, state.w * v.z, state.h * v.z);
  g.globalAlpha = FOG_ALPHA;
  g.drawImage(mask, v.tx, v.ty, state.w * v.z, state.h * v.z);
  g.globalAlpha = 1;
}

function remask(){
  if(state.w && state.h) Fog.rasterise(state.strokes, state.w, state.h, mask);
  redraw();
}

/* Swap in a stored map: decode it, fit it, and pick its fog back up. */
async function loadMapRecord(rec){
  const raw = await createImageBitmap(rec.blob);
  const f = await getFog(rec.id);
  state.mapId = rec.id; state.name = rec.name;
  state.rotation = (f && f.rotation) || 0;
  state.strokes = (f && f.strokes) || [];
  /* Bring the picture to the orientation the saved strokes assume. */
  state.img = Fog.rotateImage(raw, state.rotation);
  state.w = state.img.width; state.h = state.img.height;
  const cv = sizeCanvas();
  state.view = Fog.fitView(state.w, state.h, cv.width, cv.height);
  remask();
  say(rec.name);
}

async function chooseMap(file){
  try{
    const bmp = await createImageBitmap(file);
```

```javascript
/* ── the stroke stack ──────────────────────────────────────────────
   Kept as three tiny pure functions so the suite can exercise the
   ordering rules without a canvas or a map. */
function applyPush(st, stroke){ st.strokes.push(stroke); st.undone.length = 0; return true; }
function applyUndo(st){
  if(!st.strokes.length) return false;
  st.undone.push(st.strokes.pop()); return true;
}
function applyRedo(st){
  if(!st.undone.length) return false;
  st.strokes.push(st.undone.pop()); return true;
}

const LAST_KEY = "waterdeep-table.last";
const stack = { strokes: state.strokes, undone: [] };

let saveT = null;

  if(!state.mapId) return;
  clearTimeout(saveT);
```

```html
function idb(){
  return new Promise(function(res, rej){
    const r = indexedDB.open(DB, 1);
    r.onupgradeneeded = function(){
      const db = r.result;
      if(!db.objectStoreNames.contains(MAPS)) db.createObjectStore(MAPS, {keyPath:"id"});

    r.onerror = function(){ rej(r.error); };

    const tx = db.transaction(MAPS, "readonly");
    const rq = tx.objectStore(MAPS).get(id);

    rq.onerror = () => rej(rq.error);
```

## Notes for whoever runs this

- **Do not put the map in a channel message.** It is 15 MB. Both pages are same-origin and share IndexedDB; the id is all that needs to travel. If you find yourself calling `postMessage` with a blob, the design has been misread.
- **Stroke coordinates are image-space, always.** Converting to canvas coordinates before storing will look correct on one screen and be wrong on the projector.
- **The DM's fog is 0.78 alpha; the projector's is opaque.** Getting these the same way round is a spoiler, not a cosmetic bug.
- **A zero-length line must still paint a dot** — `rasterise` special-cases it, because stroking a zero-length path draws nothing.
